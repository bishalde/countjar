# public counter: no token needed
curl {{ site_url }}/api/v2/acme/visits/up

# private counter: send a workspace token
curl -H "Authorization: Bearer cj_..." \
     "{{ site_url }}/api/v2/acme/signups/up?by=5&plan=pro"

{"name":"signups","count":1285,"counted":true,...}


const res = await fetch(
  "{{ site_url }}/api/v2/acme/visits/up"
);
const { count } = await res.json();
document.querySelector("#views").textContent = count;


import requests

r = requests.post(
    "{{ site_url }}/api/v2/acme/builds/up",
    headers={"Authorization": "Bearer cj_..."},
)
print(r.json()["count"])
